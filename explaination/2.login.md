## Get Request Data
```php

function readRequestData(): array
{
    $contentType = $_SERVER['CONTENT_TYPE'] ?? '';

    if (str_contains($contentType, 'application/json')) {
        $rawBody = file_get_contents('php://input');
        $jsonData = json_decode($rawBody ?: '', true);

        if (is_array($jsonData)) {
            return $jsonData;
        }
    }

    return $_POST;
}
```
## set a variable and store the request data inside
```php
$data = readRequestData();
```
## Set the email and password variable and check either is empty
```php
$email = strtolower(trim((string) ($data['email'] ?? '')));
$password = (string) ($data['password'] ?? '');

if ($email === '' || $password === '') {
    respond(422, [
        'success' => false,
        'message' => 'Email and password are required',
    ]);
}
```

## Select Collection users from the database, and find the user from that collection by matching the email
```php
$users = $db->selectCollection('users');
$user = $users->findOne(['email' => $email]);

// Password hashes are verified server-side; the stored password is never returned.
if ($user === null || !isset($user['password']) || !password_verify($password, (string) $user['password'])) {
    respond(401, [
        'success' => false,
        'message' => 'Invalid email or password',
    ]);
}

```

## Set Session cookie
```php
session_regenerate_id(true);

$_SESSION['user_id'] = (string) $user['user_id'];
$_SESSION['username'] = (string) $user['username'];
$_SESSION['role'] = (string) $user['role'];

respond(200, [
    'success' => true,
    'message' => 'Logged in successfully',
    'user' => [
        'user_id' => (string) $user['user_id'],
        'username' => (string) $user['username'],
        'email' => (string) $user['email'],
        'role' => (string) $user['role'],
        'anime_interest' => $user['anime_interest'] ?? '',
        'telephone' => $user['telephone'] ?? null,
    ],
]);
```
